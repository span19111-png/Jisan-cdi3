-- UNIVERSAL TOUCH & MOUSE DRAGGABLE HELPER
local function makeDraggable(handle, target)
    local dragging = false
    local dragStart = nil
    local startPos = nil

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = target.Position
        end
    end)

    handle.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            target.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
end

-- LIVE FPS COUNTER HELPER
local function attachFpsCounter(label)
    local lastTime = tick()
    local frames = 0
    RunService.RenderStepped:Connect(function()
        frames = frames + 1
        if tick() - lastTime >= 1 then
            label.Text = tostring(frames) .. " FPS"
            frames = 0
            lastTime = tick()
        end
    end)
end

-- PULSING GLOW ANIMATION HELPER
local function startGlowAnimation(stroke, minColor, maxColor, speed)
    task.spawn(function()
        local tweenInfo = TweenInfo.new(speed or 1.4, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true)
        local tween = TweenService:Create(stroke, tweenInfo, {
            Color = maxColor or Color3.fromRGB(255, 65, 80)
        })
        stroke.Color = minColor or Color3.fromRGB(160, 25, 35)
        tween:Play()
    end)
end

local function showBanner(msg, isError)
    bannerText.Text = msg
    if isError then
        bannerText.TextColor3 = Color3.fromRGB(255, 90, 90)
    else
        bannerText.TextColor3 = Color3.fromRGB(144, 238, 144)
    end

    banner.Visible = true
    if bannerTween then bannerTween:Cancel() end

    bannerTween = TweenService:Create(banner, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Position = UDim2.new(0.5, -140, 0, 16)
    })
    bannerTween:Play()

    task.delay(3, function()
        if banner.Position.Y.Offset > 0 then
            local tweenOut = TweenService:Create(banner, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
                Position = UDim2.new(0.5, -140, 0, -50)
            })
            tweenOut:Play()
            tweenOut.Completed:Connect(function()
                if banner.Position.Y.Offset <= -50 then
                    banner.Visible = false
                end
            end)
        end
    end)
end

-- RECURSION-SAFE TEXT OBSERVER
-- ========================================================
local lastFruitName = ""
local statusMode = "none"
local KEYWORD = "fruit added to"
local isModifyingText = false

local function watchObject(obj)
    if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
        local function checkText()
            if isModifyingText then return end
            local text = string.lower(obj.Text)

            if string.find(text, KEYWORD) or string.find(text, "spawned successfully") or string.find(text, "not avalible") then
                isModifyingText = true
                if statusMode == "error" then
                    local target = "Dragon (west) Is Currently not Avalible."
                    if obj.Text ~= target then
                        obj.Text = target
                        obj.TextColor3 = Color3.fromRGB(255, 80, 80)
                    end
                elseif statusMode == "spawn" then
                    local target = lastFruitName .. " Fruit spawned successfully!"
                    if obj.Text ~= target then
                        obj.Text = target
                        obj.TextColor3 = Color3.fromRGB(144, 238, 144)
                    end
                end
                isModifyingText = false
            end
        end
        checkText()
        obj:GetPropertyChangedSignal("Text"):Connect(checkText)
    end
end

for _, obj in ipairs(playerGui:GetDescendants()) do watchObject(obj) end
playerGui.DescendantAdded:Connect(watchObject)

local currentTarget = nil

local function getPartnerFromTradeSeat()
    local char = player.Character
    if not char then return nil end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid or not humanoid.SeatPart then return nil end

    local mySeat = humanoid.SeatPart
    local tableModel = mySeat:FindFirstAncestorOfClass("Model")
    if not tableModel then return nil end

    for _, seat in ipairs(tableModel:GetDescendants()) do
        if seat:IsA("Seat") and seat ~= mySeat and seat.Occupant then
            local oppChar = seat.Occupant.Parent
            local targetPlayer = Players:GetPlayerFromCharacter(oppChar)
            if targetPlayer and targetPlayer ~= player then
                return targetPlayer
            end
        end
    end
    return nil
end

local function getPartnerFromTradeGui()
    for _, g in ipairs(playerGui:GetChildren()) do
        if g:IsA("ScreenGui") and g.Enabled then
            for _, d in ipairs(g:GetDescendants()) do
                if d:IsA("TextLabel") and string.find(string.lower(d.Text), "treasure trade") then
                    local frame = d.Parent
                    if frame then
                        for _, p in ipairs(Players:GetPlayers()) do
                            if p ~= player then
                                for _, label in ipairs(frame:GetDescendants()) do
                                    if label:IsA("TextLabel") and (label.Text == p.Name or label.Text == p.DisplayName) then
                                        return p
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return nil
end

task.spawn(function()
    while true do
        task.wait(0.5)
        local foundPartner = getPartnerFromTradeSeat() or getPartnerFromTradeGui()

        if foundPartner and foundPartner ~= currentTarget then
            currentTarget = foundPartner
            tradeLabel.Text = "Trading with:\n" .. foundPartner.DisplayName .. " (@" .. foundPartner.Name .. ")"
            
            task.spawn(function()
                local success, url = pcall(function()
                    return Players:GetUserThumbnailAsync(
                        foundPartner.UserId,
                        Enum.ThumbnailType.HeadShot,
                        Enum.ThumbnailSize.Size100x100
                    )
                end)
                if success and url then
                    tradeAvatar.Image = url
                    tradeAvatar.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
                end
            end)
        elseif not foundPartner and currentTarget ~= nil then
            currentTarget = nil
            tradeLabel.Text = "Trading with:\nNone (Sit at trade table)"
            tradeAvatar.Image = ""
            tradeAvatar.BackgroundColor3 = Color3.fromRGB(25, 28, 33)
        end
    end
end)
